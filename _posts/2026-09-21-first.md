document.addEventListener("DOMContentLoaded", () => {
  // 1. 地図の初期化 (中心の緯度経度、ズーム倍率)
  const map = L.map('blog-map').setView([34.825, 135.479], 13); // 例：箕面付近

  // 2. 地図のタイルデザイン（OpenStreetMap）を読み込み
  L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
    maxZoom: 19,
    attribution: '&copy; <a href="https://www.openstreetmap.org/copyright">OpenStreetMap</a> contributors'
  }).addTo(map);

  // 3. マーカー（ピン）の追加と色の変更（CircleMarkerを使うと色や大きさが自由自在です）
  // 行きたい場所1（赤色のピン）
  L.circleMarker([84.125, 132.579], {
    color: '#ff4d4f',      // 枠線の色
    fillColor: '#ff4d4f',  // 塗りつぶしの色
    fillOpacity: 0.8,      // 不透明度
    radius: 10             // 大きさ
  }).addTo(map).bindPopup("<b>行きたい場所1</b><br>カフェやお店など");

  // 行きたい場所2（青色のピン）
  L.circleMarker([134.535, 435.189], {
    color: '#1890ff',
    fillColor: '#1890ff',
    fillOpacity: 0.8,
    radius: 10
  }).addTo(map).bindPopup("<b>行きたい場所2</b><br>ホテルや観光地など");
});
